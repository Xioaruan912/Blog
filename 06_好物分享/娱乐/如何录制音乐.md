# Reaper使用方法



直接拖入伴奏 然后设置第二个轨道 当作录制

不要听到自己的声音 就按下这个按钮

![image-20260723231943144](https://raw.githubusercontent.com/Xioaruan912/pic/main/20260723231943162.png)

然后点击左边的红色开始监听

按Ctrl+R 进行录制 这样结束后就可以听到自己的了

如果要导出 就记得去渲染即可

# 歌词

希望可以边录制边有歌词很简单

获取带时间轴的软件：https://github.com/tranxuanthang/lrcget



新建一个文件`Auto_Import_LRC.lua`

```
-- 自动解析 LRC 歌词并导入为 REAPER 标记区 (Region)
function parse_lrc_time(time_str)
    local min, sec, ms = time_str:match("(%d+):(%d+)%.(%d+)")
    if not min then
        min, sec = time_str:match("(%d+):(%d+)")
        ms = "0"
    end
    if min and sec then
        ms = ms .. string.rep("0", 3 - #ms)
        return tonumber(min) * 60 + tonumber(sec) + tonumber(ms) / 1000
    end
    return nil
end

local retval, filename = reaper.GetUserFileNameForRead("", "选择 LRC 歌词文件", "lrc")
if retval then
    local file = io.open(filename, "r")
    if not file then return end
    
    local lyrics = {}
    for line in file:lines() do
        local time_str, text = line:match("%[(%d+:%d+%.?%d*)%](.*)")
        if time_str and text and text:gsub("%s+", "") ~= "" then
            local time_sec = parse_lrc_time(time_str)
            if time_sec then
                table.insert(lyrics, {time = time_sec, text = text})
            end
        end
    end
    file:close()

    reaper.Undo_BeginBlock()
    for i = 1, #lyrics do
        local start_time = lyrics[i].time
        local end_time = (i < #lyrics) and lyrics[i+1].time or (start_time + 3)
        reaper.AddProjectMarker2(0, true, start_time, end_time, lyrics[i].text, -1, 0)
    end
    reaper.Undo_EndBlock("自动导入 LRC 歌词", -1)
    reaper.UpdateTimeline()
end
```
